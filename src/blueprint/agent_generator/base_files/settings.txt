# idempotency_enabled = true
# idempotency_ttl = 1500

# Required as soon as a scheduler is registered -- it has no default, because neither
# value is safe to inherit. "event" starts no in-process timer: the tick arrives as an
# event on <app_name>.scheduler.<scheduler_name>, published by an external CronJob, so
# exactly one replica runs it -- and it needs event_bus set. "in_process" runs an
# APScheduler timer in the process and claims each tick in the cache so one replica
# runs it; it is right for local development and plain Docker, and it needs
# .with_cache() or every replica runs every tick.
scheduler_mode = "in_process"

# Set this to publish events without consuming any -- a scheduler that reports what

# Logging and telemetry. These are flat keys, not sections: the framework reads
# log_level, log_format, otel_enabled and token_metrics_enabled from the top level
# of the environment table, so a [default.logging] or [default.observability]
# section would be read by nothing at all.
#
# log_level and log_format describe the process. If this project is later hosted as
# one agent of a group, the group's own settings file decides both and these two are
# reported and ignored -- one process configures logging once.
